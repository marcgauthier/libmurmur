cmake_minimum_required(VERSION 3.21)
project(libmurmur VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  message(FATAL_ERROR "Windows builds require MinGW-w64 with the POSIX thread model")
endif()
option(MURMUR_PLATFORM_TESTS_ONLY "Build dependency-free portability tests only" OFF)
include(cmake/platform_tests.cmake)
if(MURMUR_PLATFORM_TESTS_ONLY)
  return()
endif()
find_package(Threads REQUIRED)
option(MURMUR_COMPILE_CHECK_ONLY "Compile library sources without linking or bundling dependencies" OFF)
option(MURMUR_CRSQL_TESTS_ONLY "Build SQLite/cr-sqlite integration tests without RocksDB" OFF)
if(NOT MURMUR_COMPILE_CHECK_ONLY)
  include(cmake/crsqlite.cmake)
endif()
if(MURMUR_CRSQL_TESTS_ONLY)
  return()
endif()
if(NOT MURMUR_COMPILE_CHECK_ONLY)
  set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
  set(BUILD_TOOL OFF CACHE BOOL "" FORCE)
  set(BUILD_LIBSSL OFF CACHE BOOL "" FORCE)
  set(DISABLE_GO ON CACHE BOOL "" FORCE)
  set(DISABLE_PERL ON CACHE BOOL "" FORCE)
  add_subdirectory(deps/aws-lc EXCLUDE_FROM_ALL)
  set(WITH_TESTS OFF CACHE BOOL "" FORCE)
  set(WITH_BENCHMARK_TOOLS OFF CACHE BOOL "" FORCE)
  set(WITH_TOOLS OFF CACHE BOOL "" FORCE)
  set(WITH_CORE_TOOLS OFF CACHE BOOL "" FORCE)
  set(WITH_GFLAGS OFF CACHE BOOL "" FORCE)
  set(WITH_LIBURING OFF CACHE BOOL "" FORCE)
  set(PORTABLE ON CACHE BOOL "" FORCE)
  set(FAIL_ON_WARNINGS OFF CACHE BOOL "" FORCE)
  set(ROCKSDB_BUILD_SHARED OFF CACHE BOOL "" FORCE)
  set(WITH_WINDOWS_UTF8_FILENAMES ON CACHE BOOL "" FORCE)
  add_subdirectory(deps/rocksdb EXCLUDE_FROM_ALL)
endif()
file(GLOB native CONFIGURE_DEPENDS deps/murmur/*.c deps/murmur/*.cc)
list(FILTER native EXCLUDE REGEX "_selftest\\.(c|cc)$")
file(GLOB zstd CONFIGURE_DEPENDS deps/zstd/*.c)
file(GLOB tls CONFIGURE_DEPENDS deps/mbedtls/library/*.c deps/picotls/lib/*.c deps/picoquic/picoquic/*.c)
list(FILTER tls EXCLUDE REGEX "picoquic_ptls_fusion.c$")
if(MURMUR_COMPILE_CHECK_ONLY)
  set(MURMUR_LIBRARY_TYPE OBJECT)
else()
  set(MURMUR_LIBRARY_TYPE STATIC)
endif()
add_library(murmur ${MURMUR_LIBRARY_TYPE} libmurmur.c deps/sqlite/sqlite3.c deps/crsqlite/wrapper_tu.c
  deps/codec/codec.cc deps/codec/aesgcm.cc deps/keyring/keyring.cc deps/keyring/scrypt.cc
  deps/keyring/sha256.c deps/keyring/hkdf.c deps/keyring/crc32c.c
  deps/vec0/vec0.c
  deps/picoquic/picoquic_mbedtls/ptls_mbedtls.c
  deps/picoquic/picoquic_mbedtls/ptls_mbedtls_sign.c ${native} ${zstd} ${tls})
add_library(murmur::murmur ALIAS murmur)
target_include_directories(murmur PUBLIC . deps deps/sqlite deps/murmur PRIVATE
  deps/codec deps/keyring deps/zstd deps/vec0
  deps/crsqlite/include deps/aws-lc/include deps/rocksdb/include
  deps/picoquic/picoquic deps/picoquic/picoquic_mbedtls deps/picotls/include deps/mbedtls/include)
target_compile_definitions(murmur PRIVATE SQLITE_CORE SQLITE_ENABLE_FTS5 SQLITE_THREADSAFE=1
  SQLITE_ENABLE_PREUPDATE_HOOK=1 ZSTD_DISABLE_ASM=1 PTLS_WITHOUT_OPENSSL PTLS_WITHOUT_FUSION
  PICOQUIC_WITH_MBEDTLS MBEDTLS_THREADING_C MBEDTLS_THREADING_PTHREAD)
target_link_libraries(murmur PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
if(NOT MURMUR_COMPILE_CHECK_ONLY)
  target_link_libraries(murmur PUBLIC rocksdb crypto murmur-crsqlite-core)
endif()
if(WIN32)
  target_compile_definitions(murmur PRIVATE _WINDOWS NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_link_libraries(murmur PUBLIC bcrypt crypt32 ws2_32 rpcrt4 shlwapi)
else()
  target_link_libraries(murmur PUBLIC m)
endif()
if(MURMUR_COMPILE_CHECK_ONLY)
  add_library(murmur-tools-check OBJECT sqlite3-murmur-tool/admin.c examples/c/demo.c
    sqlite3-murmur-tool/tool.c)
  target_include_directories(murmur-tools-check PRIVATE . deps deps/sqlite deps/murmur
    deps/keyring deps/crsqlite/include)
  return()
endif()
add_executable(murmur-demo examples/c/demo.c)
target_include_directories(murmur-demo PRIVATE deps/keyring deps/crsqlite/include)
target_link_libraries(murmur-demo PRIVATE murmur::murmur)

# Merge rather than extract archives: dependencies contain repeated member names.
add_custom_command(TARGET murmur POST_BUILD
  COMMAND "${CMAKE_COMMAND}" "-DAR=${CMAKE_AR}" "-DARCHIVE=$<TARGET_FILE:murmur>"
    "-DROCKSDB=$<TARGET_FILE:rocksdb>" "-DCRYPTO=$<TARGET_FILE:crypto>"
    "-DCRSQL=${MURMUR_CRSQL_ARCHIVE}"
    -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/merge_archive.cmake" VERBATIM)
option(MURMUR_BUILD_SQLITE_SHELL "Build the encrypted SQLite shell (requires patch)" ON)
if(MURMUR_BUILD_SQLITE_SHELL)
  add_subdirectory(sqlite3-murmur-tool)
endif()
add_executable(murmur-test-sync deps/murmur/sync_selftest.c)
target_include_directories(murmur-test-sync PRIVATE deps/keyring deps/crsqlite/include)
target_link_libraries(murmur-test-sync PRIVATE murmur::murmur)
add_test(NAME sync COMMAND murmur-test-sync)
set_tests_properties(sync PROPERTIES WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}")
add_executable(murmur-test-crypto-policy tests/native/crypto_policy.c)
target_include_directories(murmur-test-crypto-policy PRIVATE deps/codec deps/keyring)
target_link_libraries(murmur-test-crypto-policy PRIVATE murmur::murmur)
add_test(NAME crypto-policy COMMAND murmur-test-crypto-policy)
